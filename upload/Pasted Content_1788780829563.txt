// Explicit interfaces matching your API payload contracts
interface ChatMessage {
  role: 'user' | 'assistant' | 'system';
  content: string;
}

interface UseSEOStrategistOptions {
  projectId: string;
  onSuccess?: (reply: string) => void;
  onError?: (error: Error) => void;
}

export function useSEOStrategist({ projectId, onSuccess, onError }: UseSEOStrategistOptions) {
  const queryClient = useQueryClient();
  const cacheKey = ['projects', projectId, 'seo-chat'];

  const mutation = useMutation<string, Error, ChatMessage[]>({
    mutationFn: async (updatedHistory: ChatMessage[]) => {
      // Structure dynamic payload to perfectly match your Zod validation schema
      const payload: ChatRouteInput = {
        projectId,
        messages: updatedHistory,
      };

      const response = await fetch('/api/ai/chat', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify(payload),
      });

      const json = await response.json();

      if (!response.ok) {
        // Flatten nested Zod validation field messages for standard runtime exceptions
        if (json.error === 'Validation failed' && json.details) {
          const fieldErrors = Object.entries(json.details)
            .map(([field, msgs]) => `${field}: ${(msgs as string[]).join(', ')}`)
            .join(' | ');
          throw new Error(`Validation Error — ${fieldErrors}`);
        }
        throw new Error(json.error || 'Failed to communicate with Asymmetrical SEO Strategist.');
      }

      return json.data;
    },
    onMutate: async (newHistory) => {
      // Cancel outbound requests to prevent racing condition updates
      await queryClient.cancelQueries({ queryKey: cacheKey });

      // Snapshot current cache state for optimistic state rollback support
      const previousMessages = queryClient.getQueryData<ChatMessage[]>(cacheKey) || [];

      // Update the cache stream immediately with user intent parameters 
      queryClient.setQueryData<ChatMessage[]>(cacheKey, newHistory);

      return { previousMessages };
    },
    onSuccess: (assistantReply, variables, context) => {
      // Append the clean API response message array safely back to state cache
      queryClient.setQueryData<ChatMessage[]>(cacheKey, [
        ...variables,
        { role: 'assistant', content: assistantReply }
      ]);
      
      if (onSuccess) onSuccess(assistantReply);
    },
    onError: (err, variables, context) => {
      // Rollback to cached state structure if the validation middleware yields failures
      if (context?.previousMessages) {
        queryClient.setQueryData(cacheKey, context.previousMessages);
      }
      
      if (onError) onError(err);
    },
  });

  return {
    sendMessage: mutation.mutate,
    isLoading: mutation.isPending,
    isError: mutation.isError,
    error: mutation.error,
    chatHistory: queryClient.getQueryData<ChatMessage[]>(cacheKey) || [],
  };
}

// /app/api/ai/chat/route.ts
import { ChatRouteSchema } from '@/lib/validations/chat';
import { ASYMMETRICAL_SEO_STRATEGIST_PROMPT } from '@/ai/personas/seo-strategist';
import { zAI } from 'z-ai-web-dev-sdk';

export async function POST(req: Request) {
  try {
    const rawBody = await req.json();
    const validation = ChatRouteSchema.safeParse(rawBody);
    
    if (!validation.success) {
      return new Response(JSON.stringify({ error: 'Validation failed', details: validation.error.flatten().fieldErrors }), {
        status: 400,
        headers: { 'Content-Type': 'application/json' },
      });
    }

    const { messages, projectId } = validation.data;
    const finalizedMessages = [
      { role: 'system', content: ASYMMETRICAL_SEO_STRATEGIST_PROMPT },
      ...messages
    ];

    // Request a streaming completion from the z-ai SDK
    const responseStream = await zAI.chat.completions.create({
      model: 'preset-optimized-local-seo',
      messages: finalizedMessages,
      temperature: 0.3,
      stream: true, // Native text/event-stream activation
      metadata: { projectId }
    });

    // Pipe the downstream model blocks directly into the response runtime
    const stream = new ReadableStream({
      async start(controller) {
        const encoder = new TextEncoder();
        try {
          for await (const chunk of responseStream) {
            const content = chunk.choices[0]?.delta?.content || '';
            if (content) {
              // Wrap payload chunks inside a structured SSE data block
              controller.enqueue(encoder.encode(`data: ${JSON.stringify({ content })}\n\n`));
            }
          }
        } catch (err) {
          controller.error(err);
        } finally {
          controller.close();
        }
      },
    });

    return new Response(stream, {
      headers: {
        'Content-Type': 'text/event-stream',
        'Cache-Control': 'no-cache, no-transform',
        'Connection': 'keep-alive',
      },

/**
 * Safely parses streaming text context containing mixed markdown syntax and raw URLs.
 * Handles split or incomplete markdown tokens gracefully during SSE transit hooks.
 */
export function parseStreamingContent(text: string): string {
  if (!text) return '';

  // Regex to isolate true URLs while ignoring standard punctuation or closing parentheses
  const urlRegex = /(https?:\/\/[^\s<]+[^.,:;"')\s])/g;

  let processed = text;

  // 1. Convert raw URLs into clickable anchor elements if they aren't already wrapped in markdown markdown syntax
  processed = processed.replace(urlRegex, (url) => {
    // Avoid double-wrapping matches that are already part of an explicit markdown statement [text](url)
    const index = text.indexOf(url);
    if (index > 0 && text[index - 1] === '(') {
      return url;
    }
    return `<a href="${url}" target="_blank" rel="noopener noreferrer" class="text-terracotta underline font-semibold hover:text-clay transition-colors">${url}</a>`;
  });

  // 2. Safely parse bold markers (**bold**), supporting open-ended streaming chunks safely
  const boldParts = processed.split('**');
  if (boldParts.length > 1) {
    processed = boldParts.reduce((acc, part, i) => {
      // Every odd index represents wrapped text. If it is trailing and unclosed, treat it as normal text.
      if (i % 2 !== 0) {
        const hasClosing = i < boldParts.length - 1;
        return acc + (hasClosing ? `<strong class="font-bold text-forest">${part}</strong>` : `**${part}`);
      }
      return acc + part;
    }, '');
  }

  // 3. Process simple bullet alignments cleanly
  processed = processed.split('\n').map(line => {
    if (line.trim().startsWith('- ')) {
      return `<li class="list-disc list-inside ml-2 my-1 text-bark/90">${line.trim().substring(2)}</li>`;
    }
    return line;
  }).join('\n');

  return processed;
}

      metadata: { projectId }
